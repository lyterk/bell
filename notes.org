#+begin_src rust
use std::fs;
use std::thread;
use std::time::Duration;
use rand::Rng;
use rand_distr::{Exp, Distribution};
use rodio::{Decoder, OutputStream, Sink};
use chrono::Local;

fn is_quiet_hours() -> bool {
    let hour = Local::now().hour();
    hour >= 23 || hour < 7
}

fn play_random_file() -> Result<(), Box<dyn std::error::Error>> {
    let entries: Vec<_> = fs::read_dir("resources/")?
        .filter_map(|e| e.ok())
        .filter(|e| e.path().is_file())
        .collect();

    if entries.is_empty() {
        eprintln!("No files found in resources/");
        return Ok(());
    }

    let idx = rand::rng().random_range(0..entries.len());
    let path = entries[idx].path();

    let (_stream, stream_handle) = OutputStream::try_default()?;
    let sink = Sink::try_new(&stream_handle)?;

    let file = fs::File::open(&path)?;
    let source = Decoder::new(std::io::BufReader::new(file))?;
    sink.append(source);
    sink.sleep_until_end();

    println!("Played: {:?}", path);
    Ok(())
}

fn main() {
    let exp = Exp::new(1.0f64).expect("Failed to create exponential distribution");
    let thirty_minutes_secs = 30.0 * 60.0;

    loop {
        let sample = exp.sample(&mut rand::rng());
        let wait_secs = (sample * thirty_minutes_secs) as u64;

        println!("Next bell in {} seconds ({:.1} minutes)", wait_secs, wait_secs as f64 / 60.0);
        thread::sleep(Duration::from_secs(wait_secs));

        if is_quiet_hours() {
            println!("Quiet hours (11pm–7am), skipping bell.");
        } else {
            if let Err(e) = play_random_file() {
                eprintln!("Error playing file: {}", e);
            }
        }
    }
}
#+end_src

Yes, you can reference the package's resources directory directly using the Nix store path. In your NixOS module, you'd set =resourcesDir= to ="${bell.packages.${system}.default}/resources"= (or similar), which points to the =resources/= directory as it's bundled in the package.

For this to work, the =resources/= directory needs to be included when building the package (i.e., copied in your =installPhase= or via =src=). If your =flake.nix= builds with something like:

#+begin_src nix
installPhase = ''
  install -Dm755 target/release/bell $out/bin/bell
  cp -r resources $out/resources
'';
#+end_src

Then you can reference it as:

#+begin_src nix
services.bell = {
  enable = true;
  resourcesDir = "${bell.packages.x86_64-linux.default}/resources";
};
#+end_src

This avoids hardcoding =/var/lib/bell= and uses whatever is in the repo.

Or run as your regular user instead of a dedicated system user, which simplifies audio access considerably.

Looking at the rodio 0.22.2 docs, the API has changed significantly. =OutputStream=, =Sink= are replaced with =DeviceSinkBuilder=, =Player=, etc.

#+begin_src rust
use chrono::{Local, Timelike};
use rand::RngExt;
use rand_distr::{Distribution, Exp};
use rodio::{Decoder, Player};
use std::fs;
use std::thread;
use std::time::Duration;

fn is_quiet_hours() -> bool {
    let hour = Local::now().hour();
    hour >= 23 || hour < 7
}

fn play_random_file() -> Result<(), Box<dyn std::error::Error>> {
    let entries: Vec<_> = fs::read_dir("resources/")?
        .filter_map(|e| e.ok())
        .filter(|e| e.path().is_file())
        .collect();

    if entries.is_empty() {
        eprintln!("No files found in resources/");
        return Ok(());
    }

    let idx = rand::rng().random_range(0..entries.len());
    let path = entries[idx].path();

    let handle = rodio::DeviceSinkBuilder::open_default_sink()?;
    let player = Player::connect_new(&handle.mixer());

    let file = fs::File::open(&path)?;
    let source = Decoder::try_from(std::io::BufReader::new(file))?;
    player.append(source);
    player.sleep_until_end();

    println!("Played: {:?}", path);
    Ok(())
}

fn main() {
    let exp = Exp::new(1.0f64).expect("Failed to create exponential distribution");
    let thirty_minutes_secs = 50.0 * 60.0;

    loop {
        let sample = exp.sample(&mut rand::rng());
        let wait_secs = (sample * thirty_minutes_secs) as u64;

        println!(
            "Next bell in {} seconds ({:.1} minutes)",
            wait_secs,
            wait_secs as f64 / 60.0
        );
        thread::sleep(Duration::from_secs(wait_secs));

        if is_quiet_hours() {
            println!("Quiet hours (11pm–7am), skipping bell.");
        } else {
            if let Err(e) = play_random_file() {
                eprintln!("Error playing file: {}", e);
            }
        }
    }
}
#+end_src

Key changes:
- =OutputStream::try_default()= → =DeviceSinkBuilder::open_default_sink()=
- =Sink::try_new(&stream_handle)= → =Player::connect_new(&handle.mixer())=
- =Decoder::new(...)= → =Decoder::try_from(...)=
- =sink.append= / =sink.sleep_until_end= → =player.append= / =player.sleep_until_end
